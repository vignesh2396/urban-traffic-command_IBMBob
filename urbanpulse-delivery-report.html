🚀 Running the App

# Clone
git clone https://github.com/vignesh2396/urban-traffic-command_IBMBob.git
cd urban-traffic-command_IBMBob

# Install
pip install -r requirements.txt

# Configure (add GEMINI_API_KEY)
cp .env.example .env

# Seed DB + RAG
python -m data.seed_data
python -m rag.embeddings

# Launch
streamlit run app.py
# → http://localhost:8501

# Run all tests
pytest tests/ -v         # 24/24 PASS

# Generate video
python create_video.py   # → UrbanPulse_Demo.mp4 (80s)
